# Configurar um novo projeto é bastante simples com a CLI do Nest
npm i -g @nestjs/cli
nest new project-name
cd project-name

# instalando dependências
npm i @nestjs/jwt jsonwebtoken @nestjs/mongoose mongoose bcrypt class-transformer class-validator @nestjs/passport passport passport-jwt uuid dotenv
npm install mongodb
npm i -D @types/bcrypt


# criando estruturas
nest g module users
nest g controller users --no-spec
nest g service users --no-spec

nest g module auth
nest g service auth
